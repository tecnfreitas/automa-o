function Write-Header {
    param(
        [string]$Text
    )

    Write-Host ""
    Write-Host "============================================================" -ForegroundColor Cyan
    Write-Host " $Text" -ForegroundColor Cyan
    Write-Host "============================================================" -ForegroundColor Cyan
}

function Write-OK {
    param(
        [string]$Text
    )

    Write-Host "[OK] $Text" -ForegroundColor Green
}

function Write-WarningMessage {
    param(
        [string]$Text
    )

    Write-Host "[!] $Text" -ForegroundColor Yellow
}

function Write-Risk {
    param(
        [string]$Text
    )

    Write-Host "[RISCO] $Text" -ForegroundColor Red
}

function Get-RiskLevel {

    param(
        [int]$Score
    )

    if ($Score -ge 70) {
        return "ALTO"
    }

    if ($Score -ge 40) {
        return "MÉDIO"
    }

    if ($Score -ge 20) {
        return "ATENÇÃO"
    }

    return "BAIXO"
}

function Get-RiskColor {

    param(
        [string]$Risk
    )

    switch ($Risk) {

        "ALTO" {
            return "Red"
        }

        "MÉDIO" {
            return "DarkYellow"
        }

        "ATENÇÃO" {
            return "Yellow"
        }

        default {
            return "Green"
        }
    }
}

function Test-SuspiciousText {

    param(
        [string]$Text
    )

    $Matches = New-Object System.Collections.Generic.List[string]

    if ([string]::IsNullOrWhiteSpace($Text)) {
        return $Matches
    }

    $LowerText = $Text.ToLower()

    foreach ($Keyword in $SuspiciousKeywords) {

        if ($LowerText.Contains($Keyword.ToLower())) {

            if (-not $Matches.Contains($Keyword)) {
                $Matches.Add($Keyword)
            }
        }
    }

    return $Matches
}

function Get-SignatureStatus {

    param(
        [string]$InstallLocation
    )

    if ([string]::IsNullOrWhiteSpace($InstallLocation)) {
        return "NÃO ANALISADA"
    }

    if (-not (Test-Path $InstallLocation)) {
        return "CAMINHO NÃO ENCONTRADO"
    }

    try {

        $Executables = Get-ChildItem `
            -Path $InstallLocation `
            -Filter "*.exe" `
            -File `
            -Recurse `
            -ErrorAction SilentlyContinue |
            Select-Object -First 10

        if (-not $Executables) {
            return "NENHUM EXE ENCONTRADO"
        }

        $Unsigned = 0
        $Valid = 0
        $Unknown = 0

        foreach ($Exe in $Executables) {

            $Signature = Get-AuthenticodeSignature `
                -FilePath $Exe.FullName `
                -ErrorAction SilentlyContinue

            switch ($Signature.Status) {

                "Valid" {
                    $Valid++
                }

                "NotSigned" {
                    $Unsigned++
                }

                default {
                    $Unknown++
                }
            }
        }

        if ($Unsigned -gt 0 -and $Valid -eq 0) {
            return "SEM ASSINATURA"
        }

        if ($Unsigned -gt 0) {
            return "MISTA"
        }

        if ($Valid -gt 0) {
            return "ASSINADA"
        }

        return "NÃO CONFIRMADA"

    }
    catch {

        return "ERRO NA ANÁLISE"
    }
}

Write-Header "1. INVENTÁRIO DE SOFTWARE"

$RegistryPaths = @(
    "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*",
    "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall\*",
    "HKCU:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\*"
)

$RegistrySoftware = foreach ($Path in $RegistryPaths) {

    Get-ItemProperty $Path -ErrorAction SilentlyContinue |
    Where-Object {
        -not [string]::IsNullOrWhiteSpace($_.DisplayName)
    }
}

$UniqueRegistrySoftware = $RegistrySoftware |
    Sort-Object DisplayName, DisplayVersion -Unique

foreach ($App in $UniqueRegistrySoftware) {

    $Name = $App.DisplayName
    $Publisher = $App.Publisher
    $VersionApp = $App.DisplayVersion
    $InstallDate = $App.InstallDate
    $InstallLocation = $App.InstallLocation

    if ([string]::IsNullOrWhiteSpace($InstallLocation)) {

        $UninstallString = $App.UninstallString

        if ($UninstallString -and
            $UninstallString -match '([A-Za-z]:\\[^"]+)') {

            $PossiblePath = $Matches[1]

            if (Test-Path $PossiblePath) {
                $InstallLocation = Split-Path $PossiblePath -Parent
            }
        }
    }

    $TextToAnalyze = @(
        $Name
        $Publisher
        $InstallLocation
    ) -join " "

    $Indicators = Test-SuspiciousText $TextToAnalyze

    $Score = 0

    # ------------------------------------------------------------
    # Publisher
    # ------------------------------------------------------------

    if ([string]::IsNullOrWhiteSpace($Publisher)) {

        $Score += 10
    }

    # ------------------------------------------------------------
    # Indicadores de ativação/crack
    # ------------------------------------------------------------

    if ($Indicators.Count -gt 0) {

        $Score += ($Indicators.Count * 25)

    }

    # ------------------------------------------------------------
    # Caminho suspeito
    # ------------------------------------------------------------

    $LowerPath = $InstallLocation.ToLower()

    foreach ($SuspiciousPath in $SuspiciousPaths) {

        if ($LowerPath.Contains($SuspiciousPath.ToLower())) {

            $Score += 30

            break
        }
    }

    # ------------------------------------------------------------
    # Caminho em AppData / Temp / Public
    # ------------------------------------------------------------

    if ($LowerPath -match "\\appdata\\local\\temp\\") {

        $Score += 15
    }

    if ($LowerPath -match "\\users\\public\\") {

        $Score += 15
    }

    # ------------------------------------------------------------
    # Assinatura
    # ------------------------------------------------------------

    $Signature = Get-SignatureStatus $InstallLocation

    if ($Signature -eq "SEM ASSINATURA") {

        $Score += 20
    }

    # ------------------------------------------------------------
    # Limite
    # ------------------------------------------------------------

    if ($Score -gt 100) {
        $Score = 100
    }

    $Risk = Get-RiskLevel $Score

    $SoftwareList.Add(
        [PSCustomObject]@{
            Nome = $Name
            Publisher = $Publisher
            Versao = $VersionApp
            DataInstalacao = $InstallDate
            Caminho = $InstallLocation
            Assinatura = $Signature
            Indicadores = ($Indicators -join ", ")
            Score = $Score
            Risco = $Risk
            Origem = "Registro"
        }
    )

    if ($Score -ge 40) {

        $GlobalFindings.Add(
            "$Name - risco $Risk - Score $Score"
        )
    }
}

Write-OK "$($SoftwareList.Count) softwares encontrados através do Registro."

# ================================================================

Write-Header "3. PROCURA DE INDICADORES SUSPEITOS"

$SearchLocations = @(
    "$env:ProgramFiles",
    "${env:ProgramFiles(x86)}",
    "$env:LOCALAPPDATA",
    "$env:APPDATA"
)

$SuspiciousFiles = New-Object System.Collections.Generic.List[object]

foreach ($Location in $SearchLocations) {

    if (-not (Test-Path $Location)) {
        continue
    }

    Write-Info "Analisando: $Location"

    try {

        $Files = Get-ChildItem `
            -Path $Location `
            -File `
            -Recurse `
            -ErrorAction SilentlyContinue |
            Where-Object {
                $_.Name -match "(?i)crack|keygen|activator|kmspico|kmsauto|autokms|loader|patcher|license.?bypass"
            } |
            Select-Object -First 100

        foreach ($File in $Files) {

            $Indicators = Test-SuspiciousText $File.FullName

            $Signature = Get-AuthenticodeSignature `
                -FilePath $File.FullName `
                -ErrorAction SilentlyContinue

            $SignatureStatus = $Signature.Status

            $SuspiciousFiles.Add(
                [PSCustomObject]@{
                    Arquivo = $File.Name
                    Caminho = $File.FullName
                    Indicadores = ($Indicators -join ", ")
                    Assinatura = $SignatureStatus
                }
            )
        }
    }
    catch {}
}

if ($SuspiciousFiles.Count -gt 0) {

    Write-Risk "$($SuspiciousFiles.Count) arquivo(s) com nomes potencialmente suspeitos encontrados."

    $GlobalFindings.Add(
        "$($SuspiciousFiles.Count) arquivos apresentaram nomes associados a ferramentas de ativação/crack."

}

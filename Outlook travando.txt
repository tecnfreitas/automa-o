$ErrorActionPreference = "SilentlyContinue"

$BaseFolder = "C:\IT\Outlook-Freeze-Diagnostic"

$TimeStamp = Get-Date -Format "yyyy-MM-dd_HH-mm-ss"

$LogFile = Join-Path $BaseFolder "Outlook-Diagnostic-$TimeStamp.log"

$ReportFile = Join-Path $BaseFolder "Outlook-Diagnostic-$TimeStamp.html"

# VARIÁVEIS DE DIAGNÓSTICO
# ============================================================

$Issues = New-Object System.Collections.ArrayList
$Warnings = New-Object System.Collections.ArrayList
$Successes = New-Object System.Collections.ArrayList
$Actions = New-Object System.Collections.ArrayList

$Global:OutlookWasRunning = $false
$Global:RepairMode = 0

# ============================================================
# FUNÇÃO LOG
# ============================================================

function Write-Log {

    param(
        [string]$Message,
        [ValidateSet(
            "INFO",
            "OK",
            "WARNING",
            "ERROR",
            "ACTION"
        )]
        [string]$Level = "INFO"
    )

    $Time = Get-Date -Format "HH:mm:ss"

    switch ($Level) {

        "INFO" {
            $Color = "White"
        }

        "OK" {
            $Color = "Green"
        }

        "WARNING" {
            $Color = "Yellow"
        }

        "ERROR" {
            $Color = "Red"
        }

        "ACTION" {
            $Color = "Cyan"
        }
    }

    $Line = "[$Time] [$Level] $Message"

    Write-Host $Line -ForegroundColor $Color

    Add-Content -Path $LogFile -Value $Line
}

# ============================================================
# FUNÇÃO SEÇÃO
# ============================================================

function Write-Section {

    param(
        [string]$Title
    )

    Write-Host ""
    Write-Host "============================================================" -ForegroundColor Cyan
    Write-Host $Title -ForegroundColor Cyan
    Write-Host "============================================================" -ForegroundColor Cyan

    Add-Content -Path $LogFile -Value ""
    Add-Content -Path $LogFile -Value "============================================================"
    Add-Content -Path $LogFile -Value $Title
    Add-Content -Path $LogFile -Value "============================================================"
}

# ============================================================
# FUNÇÕES DE RESULTADO
# ============================================================

function Add-Issue {

    param(
        [string]$Message
    )

    [void]$Issues.Add($Message)

    Write-Log $Message "ERROR"
}

function Add-Warning {

    param(
        [string]$Message
    )

    [void]$Warnings.Add($Message)

    Write-Log $Message "WARNING"
}

function Add-Success {

    param(
        [string]$Message
    )

    [void]$Successes.Add($Message)

    Write-Log $Message "OK"
}

function Add-Action {

    param(
        [string]$Message
    )

    [void]$Actions.Add($Message)

    Write-Log $Message "ACTION"
}

# SISTEMA
# ============================================================

Write-Section "1. INFORMAÇÕES DO SISTEMA"

$OS = Get-CimInstance Win32_OperatingSystem

$ComputerName = $env:COMPUTERNAME

$WindowsName = $OS.Caption

$WindowsBuild = $OS.BuildNumber

$Architecture = $OS.OSArchitecture

$LastBoot = $OS.LastBootUpTime

Write-Log "Computador: $ComputerName"
Write-Log "Windows: $WindowsName"
Write-Log "Build: $WindowsBuild"
Write-Log "Arquitetura: $Architecture"
Write-Log "Último boot: $LastBoot"

# ============================================================
# CPU
# ============================================================

Write-Section "2. CPU"

try {

    $CPU = Get-Counter `
        '\Processor(_Total)\% Processor Time' `
        -SampleInterval 1 `
        -MaxSamples 2

    $CPUUsage = [math]::Round(
        ($CPU.CounterSamples |
        Select-Object -Last 1).CookedValue,
        2
    )

    Write-Log "CPU atual: $CPUUsage %"

    if ($CPUUsage -ge 90) {

        Add-Issue "CPU acima de 90% durante o diagnóstico."

    }
    elseif ($CPUUsage -ge 75) {

        Add-Warning "CPU acima de 75%."

    }
    else {

        Add-Success "CPU dentro de níveis normais."
    }

}
catch {

    Add-Warning "Não foi possível obter utilização da CPU."
}

# ============================================================
# RAM
# ============================================================

Write-Section "3. MEMÓRIA RAM"

$TotalRAM = [math]::Round(
    $OS.TotalVisibleMemorySize / 1MB,
    2
)

$FreeRAM = [math]::Round(
    $OS.FreePhysicalMemory / 1MB,
    2
)

$UsedRAM = $TotalRAM - $FreeRAM

$RAMPercent = [math]::Round(
    ($UsedRAM / $TotalRAM) * 100,
    2
)

Write-Log "RAM total: $TotalRAM GB"
Write-Log "RAM livre: $FreeRAM GB"
Write-Log "RAM utilizada: $RAMPercent %"

if ($RAMPercent -ge 90) {

    Add-Issue "Memória RAM acima de 90%."

}
elseif ($RAMPercent -ge 80) {

    Add-Warning "Memória RAM acima de 80%."

}
else {

    Add-Success "Memória RAM dentro de níveis normais."
}

# ============================================================
# DISCO
# ============================================================

Write-Section "4. DISCO C:"

$Disk = Get-CimInstance Win32_LogicalDisk `
    -Filter "DeviceID='C:'"

$DiskTotal = [math]::Round(
    $Disk.Size / 1GB,
    2
)

$DiskFree = [math]::Round(
    $Disk.FreeSpace / 1GB,
    2
)

$DiskPercent = [math]::Round(
    (($DiskTotal - $DiskFree) / $DiskTotal) * 100,
    2
)

Write-Log "Capacidade: $DiskTotal GB"
Write-Log "Livre: $DiskFree GB"
Write-Log "Utilização: $DiskPercent %"

if ($DiskFree -lt 10) {

    Add-Issue "Menos de 10 GB livres no disco C:."

}
elseif ($DiskFree -lt 20) {

    Add-Warning "Menos de 20 GB livres no disco C:."

}
else {

    Add-Success "Espaço em disco adequado."
}

# ============================================================
# OUTLOOK
# ============================================================

Write-Section "5. PROCESSO OUTLOOK"

$OutlookProcesses = Get-Process OUTLOOK `
    -ErrorAction SilentlyContinue

if ($OutlookProcesses) {

    $Global:OutlookWasRunning = $true

    Add-Success "Outlook está em execução."

    foreach ($Process in $OutlookProcesses) {

        $RAMMB = [math]::Round(
            $Process.WorkingSet64 / 1MB,
            2
        )

        Write-Log "PID: $($Process.Id)"
        Write-Log "Memória: $RAMMB MB"

        if ($RAMMB -ge 3000) {

            Add-Issue "Outlook utilizando mais de 3 GB de RAM."

        }
        elseif ($RAMMB -ge 2000) {

            Add-Warning "Outlook utilizando mais de 2 GB de RAM."
        }
    }

}
else {

    Add-Warning "Outlook não está em execução."
}

# ============================================================
# ADD-INS
# ============================================================

Write-Section "6. ADD-INS DO OUTLOOK"

$AddinPaths = @(
    "HKCU:\Software\Microsoft\Office\Outlook\Addins",
    "HKLM:\Software\Microsoft\Office\Outlook\Addins",
    "HKLM:\Software\WOW6432Node\Microsoft\Office\Outlook\Addins"
)

$Addins = @()

foreach ($Path in $AddinPaths) {

    if (Test-Path $Path) {

        $Items = Get-ChildItem $Path

        foreach ($Item in $Items) {

            $Properties = Get-ItemProperty `
                $Item.PSPath

            $Name = $Item.PSChildName

            $LoadBehavior = $Properties.LoadBehavior

            $Addins += [PSCustomObject]@{

                Name = $Name
                LoadBehavior = $LoadBehavior
                Path = $Path
            }

            Write-Log "Add-in: $Name"
            Write-Log "LoadBehavior: $LoadBehavior"
        }
    }
}

if ($Addins.Count -eq 0) {

    Add-Success "Nenhum Add-in COM detectado."

}
else {

    Add-Warning "$($Addins.Count) Add-in(s) encontrados."

    foreach ($Addin in $Addins) {

        if ($Addin.LoadBehavior -eq 3) {

            Write-Log "Add-in carregado automaticamente: $($Addin.Name)"
        }
    }
}

# ============================================================
# OFFICE
# ============================================================

Write-Section "7. MICROSOFT OFFICE / MICROSOFT 365"

$OfficeInstalled = $false

$OfficeRegistryPaths = @(
    "HKLM:\SOFTWARE\Microsoft\Office\ClickToRun\Configuration",
    "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Office\ClickToRun\Configuration"
)

foreach ($Path in $OfficeRegistryPaths) {

    if (Test-Path $Path) {

        $OfficeInstalled = $true

        $OfficeConfig = Get-ItemProperty $Path

        Write-Log "Microsoft 365 / Click-to-Run detectado." "OK"

        if ($OfficeConfig.VersionToReport) {

            Write-Log "Versão: $($OfficeConfig.VersionToReport)"
        }

        if ($OfficeConfig.ProductReleaseIds) {

            Write-Log "Produto: $($OfficeConfig.ProductReleaseIds)"
        }
    }
}

if (-not $OfficeInstalled) {

    Add-Warning "Microsoft 365 Click-to-Run não detectado."
}

# ============================================================
# WEBVIEW2
# ============================================================

Write-Section "8. MICROSOFT EDGE WEBVIEW2"

$WebViewRegistry = @(
    "HKLM:\SOFTWARE\Microsoft\EdgeUpdate\Clients",
    "HKLM:\SOFTWARE\WOW6432Node\Microsoft\EdgeUpdate\Clients"
)

$WebViewFound = $false

foreach ($Path in $WebViewRegistry) {

    if (Test-Path $Path) {

        $Clients = Get-ChildItem $Path

        foreach ($Client in $Clients) {

            $Properties = Get-ItemProperty $Client.PSPath

            if (
                $Properties.name -match
                "WebView2"
            ) {

                $WebViewFound = $true

                Write-Log "WebView2 detectado."
                Write-Log "Versão: $($Properties.pv)"
            }
        }
    }
}

if ($WebViewFound) {

    Add-Success "Microsoft Edge WebView2 detectado."

}
else {

    Add-Issue "Microsoft Edge WebView2 não foi detectado."
}

# ============================================================
# OST / PST
# ============================================================

Write-Section "9. ARQUIVOS OST / PST"

$OutlookPaths = @(
    "$env:LOCALAPPDATA\Microsoft\Outlook",
    "$env:USERPROFILE\Documents\Outlook Files"
)

$DataFiles = @()

foreach ($Path in $OutlookPaths) {

    if (Test-Path $Path) {

        $Files = Get-ChildItem `
            -Path $Path `
            -Recurse `
            -Include *.ost,*.pst `
            -File

        foreach ($File in $Files) {

            $SizeGB = [math]::Round(
                $File.Length / 1GB,
                2
            )

            $DataFiles += [PSCustomObject]@{

                Name = $File.Name
                Path = $File.FullName
                SizeGB = $SizeGB
            }

            Write-Log "$($File.Name) | $SizeGB GB"

            if ($SizeGB -ge 20) {

                Add-Warning `
                    "Arquivo de dados grande: $($File.Name) ($SizeGB GB)."
            }
        }
    }
}

if ($DataFiles.Count -eq 0) {

    Add-Warning "Nenhum OST/PST encontrado nos caminhos padrão."

}
else {

    Add-Success "$($DataFiles.Count) arquivo(s) OST/PST encontrado(s)."
}

# ============================================================
# CACHE
# ============================================================

Write-Section "10. CACHE DO OUTLOOK"

$OutlookCachePath =
    "$env:LOCALAPPDATA\Microsoft\Outlook"

if (Test-Path $OutlookCachePath) {

    try {

        $CacheSize = (
            Get-ChildItem `
                $OutlookCachePath `
                -Recurse `
                -File |
            Measure-Object Length -Sum
        ).Sum

        $CacheGB = [math]::Round(
            $CacheSize / 1GB,
            2
        )

        Write-Log "Tamanho do diretório: $CacheGB GB"

        Add-Success "Cache do Outlook analisado."

    }
    catch {

        Add-Warning "Não foi possível calcular o cache."
    }
}

    }

        Add-Warning "Falha ao testar $Server."
    }

# ============================================================
# EVENTOS
# ============================================================

Write-Section "14. EVENT VIEWER"

$StartDate = (Get-Date).AddDays(-7)

try {

    $Events = Get-WinEvent `
        -FilterHashtable @{
            LogName = "Application"
            StartTime = $StartDate
        } `
        -ErrorAction SilentlyContinue |
        Where-Object {

            $_.ProviderName -match `
            "Outlook|Office|Microsoft Office|Application Error"

        } |
        Select-Object -First 30

    if ($Events) {

        Add-Warning `
            "$($Events.Count) evento(s) relacionados encontrados."

        foreach ($Event in $Events) {

            Write-Log "Data: $($Event.TimeCreated)"
            Write-Log "Fonte: $($Event.ProviderName)"
            Write-Log "ID: $($Event.Id)"
            Write-Log "Mensagem: $($Event.Message)"
            Write-Log "--------------------------------------------------"
        }

    }
    else {

        Add-Success `
            "Nenhum evento relevante encontrado nos últimos 7 dias."
    }

}
catch {

    Add-Warning "Não foi possível analisar Event Viewer."
}

# ============================================================
# REPARAÇÃO SEGURA
# ============================================================

if ($Global:RepairMode -ge 2) {

    Write-Section "16. REPARAÇÃO SEGURA"

    # --------------------------------------------------------
    # TEMP
    # --------------------------------------------------------

    Add-Action "Limpando arquivos temporários."

    $TempFolders = @(
        $env:TEMP,
        "$env:LOCALAPPDATA\Temp"
    )

    foreach ($Temp in $TempFolders) {

        if (Test-Path $Temp) {

            Get-ChildItem $Temp `
                -Force `
                -ErrorAction SilentlyContinue |
                Remove-Item `
                    -Recurse `
                    -Force `
                    -ErrorAction SilentlyContinue
        }
    }

    Add-Success "Arquivos temporários processados."

    # --------------------------------------------------------
    # DNS
    # --------------------------------------------------------

    Add-Action "Limpando cache DNS."

    ipconfig /flushdns | Out-Null

    Add-Success "Cache DNS limpo."

    # --------------------------------------------------------
    # OUTLOOK
    # --------------------------------------------------------

    if ($Global:OutlookWasRunning) {

        Write-Host ""
        Write-Host "O Outlook está aberto." -ForegroundColor Yellow

        $CloseOutlook = Read-Host `
            "Deseja fechar o Outlook para aplicar reparações? (S/N)"

        if ($CloseOutlook -match "^[Ss]$") {

            Add-Action "Encerrando Outlook."

            Stop-Process `
                -Name OUTLOOK `
                -Force `
                -ErrorAction SilentlyContinue

            Start-Sleep -Seconds 3

            Add-Success "Outlook encerrado."

        }
        else {

            Add-Warning `
                "Outlook permaneceu aberto. Algumas correções foram ignoradas."
        }
    }

    # --------------------------------------------------------
    # OFFICE SERVICES
    # --------------------------------------------------------

    Add-Action "Verificando serviços do Microsoft Office."

    $OfficeServices = @(
        "ClickToRunSvc"
    )

    foreach ($ServiceName in $OfficeServices) {

        $Service = Get-Service `
            -Name $ServiceName `
            -ErrorAction SilentlyContinue

        if ($Service) {

            Write-Log `
                "$ServiceName : $($Service.Status)"

            if (
                $Service.Status -ne "Running"
            ) {

                try {

                    Start-Service `
                        -Name $ServiceName `
                        -ErrorAction Stop

                    Add-Success `
                        "$ServiceName iniciado."

                }
                catch {

                    Add-Warning `
                        "Não foi possível iniciar $ServiceName."
                }
            }
        }
    }

    # --------------------------------------------------------
    # SFC CHECK
    # --------------------------------------------------------

    Add-Action "Executando verificação SFC."

    $SFCOutput = sfc /verifyonly 2>&1

    $SFCOutput |
        Add-Content -Path $LogFile

    Add-Success "Verificação SFC concluída."

    # --------------------------------------------------------
    # DISM CHECK
    # --------------------------------------------------------

    Add-Action "Executando DISM CheckHealth."

    $DISMOutput = DISM `
        /Online `
        /Cleanup-Image `
        /CheckHealth `
        2>&1

    $DISMOutput |
        Add-Content -Path $LogFile

    Add-Success "DISM CheckHealth concluído."
}

# ============================================================
# REPARAÇÃO AVANÇADA
# ============================================================

if ($Global:RepairMode -eq 3) {

    Write-Section "17. REPARAÇÃO AVANÇADA"

    Write-Host ""
    Write-Host "Atenção:" -ForegroundColor Yellow
    Write-Host "A reparação avançada pode alterar componentes do Office."
    Write-Host "Nenhum OST/PST será apagado automaticamente."
    Write-Host ""

    $AdvancedConfirm = Read-Host `
        "Deseja continuar com a reparação avançada? (S/N)"

    if ($AdvancedConfirm -match "^[Ss]$") {

        # ----------------------------------------------------
        # OFFICE REPAIR
        # ----------------------------------------------------

        $OfficeClickToRun = @(
            "$env:ProgramFiles\Common Files\Microsoft Shared\ClickToRun\OfficeC2RClient.exe",
            "${env:ProgramFiles(x86)}\Common Files\Microsoft Shared\ClickToRun\OfficeC2RClient.exe"
        )

        $C2R = $null

        foreach ($Path in $OfficeClickToRun) {

            if (Test-Path $Path) {

                $C2R = $Path
                break
            }
        }

        if ($C2R) {

            Write-Host ""
            Write-Host "Microsoft 365 Click-to-Run detectado." -ForegroundColor Green
            Write-Host ""

            Write-Host "[1] Reparação rápida"
            Write-Host "[2] Reparação online"
            Write-Host "[0] Cancelar"

            $RepairChoice = Read-Host "Escolha"

            switch ($RepairChoice) {

                "1" {

                    Add-Action "Iniciando reparação rápida do Office."

                    Start-Process `
                        -FilePath $C2R `
                        -ArgumentList `
                        "/update user displaylevel=true forceappshutdown=true" `
                        -Wait `
                        -ErrorAction SilentlyContinue

                    Add-Success `
                        "Processo de reparação/atualização iniciado."
                }

                "2" {

                    Add-Action "Iniciando reparação online do Office."

                    Start-Process `
                        -FilePath $C2R `
                        -ArgumentList `
                        "/update user displaylevel=true forceappshutdown=true" `
                        -Wait `
                        -ErrorAction SilentlyContinue

                    Add-Success `
                        "Processo de atualização/reparação iniciado."
                }

                default {

                    Add-Warning `
                        "Reparação do Office cancelada."
                }
            }

        }
        else {

            Add-Warning `
                "OfficeC2RClient.exe não encontrado."
        }

        # ----------------------------------------------------
        # DISM RESTOREHEALTH
        # ----------------------------------------------------

        $DISMConfirm = Read-Host `
            "Deseja executar DISM RestoreHealth? (S/N)"

        if ($DISMConfirm -match "^[Ss]$") {

            Add-Action `
                "Executando DISM RestoreHealth."

            DISM `
                /Online `
                /Cleanup-Image `
                /RestoreHealth `
                2>&1 |
                Add-Content -Path $LogFile

            Add-Success `
                "DISM RestoreHealth concluído."
        }

        # ----------------------------------------------------
        # SFC
        # ----------------------------------------------------

        $SFCConfirm = Read-Host `
            "Deseja executar SFC /scannow? (S/N)"

        if ($SFCConfirm -match "^[Ss]$") {

            Add-Action `
                "Executando SFC /scannow."

            sfc /scannow 2>&1 |
                Add-Content -Path $LogFile

            Add-Success `
                "SFC /scannow concluído."
        }

    }
    else {

        Add-Warning `
            "Reparação avançada cancelada pelo usuário."
    }
}

    [void]$PossibleCauses.Add(

    [void]$PossibleCauses.Add(
